# 1. 오늘 강의 개요

GraphRAG의 핵심 단계를 직접 구현한다.

```text
corpus
  ↓ chunking
chunk별 entity/relation 추출
  ↓
graph merge
  ↓
entity resolve + weight
  ↓
community detection
  ↓
community summary
  ↓
local/global search
```

# 3. chunk별 엔티티/관계 추출

```python
EXTRACTION_PROMPT = """You are an information extraction system.
Extract entities and relations from the text as JSON.

Schema:
{
  "entities":  [{"name": "...", "type": "Person|Organisation|Product|Concept"}],
  "relations": [{"source": "...", "target": "...", "label": "..."}]
}
Output ONLY valid JSON. No commentary."""
```

```python
def strip_fence(t):
    t = t.strip()
    if t.startswith("```"):
        t = re.sub(r"^```(?:json)?\s*", "", t)
        t = re.sub(r"\s*```$", "", t)
    return t
```

```python
def extract_one_chunk(chunk) -> dict:
    msgs = [
        SystemMessage(content=EXTRACTION_PROMPT),
        HumanMessage(content=f'Text:\n"""{chunk}"""'),
    ]
    raw = llm.invoke(msgs).content
    try:
        return json.loads(strip_fence(raw))
    except json.JSONDecodeError:
        return {"entities": [], "relations": []}
```

# 4. extract_all_chunks

모든 chunk를 순회하면서 추출 결과에 chunk index를 붙인다.

```python
def extract_all_chunks(chunks) -> list[dict]:
    results = []
    for i, c in enumerate(chunks):
        r = extract_one_chunk(c)
        r["chunk_index"] = i
        results.append(r)
        print(f"chunk {i}: {len(r['entities'])} entities, {len(r['relations'])} relations")
    return results
```

# 5. merge_chunk_results

chunk별 추출 결과를 하나의 방향 그래프로 합친다.

```python
def merge_chunk_results(results: list[dict]) -> nx.DiGraph:
    G = nx.DiGraph()

    for r in results:
        ci = r["chunk_index"]

        for e in r["entities"]:
            name = e["name"]
            if name in G:
                G.nodes[name]["source_chunks"].append(ci)
            else:
                G.add_node(name, type=e.get("type", "unknown"), source_chunks=[ci])

        for rel in r["relations"]:
            s, t = rel["source"], rel["target"]
            label = rel.get("label", "")

            if s not in G:
                G.add_node(s, type="unknown", source_chunks=[ci])
            if t not in G:
                G.add_node(t, type="unknown", source_chunks=[ci])

            if G.has_edge(s, t):
                if label and label not in G[s][t]["relations"]:
                    G[s][t]["relations"].append(label)
                G[s][t]["source_chunks"].append(ci)
            else:
                G.add_edge(s, t, relations=[label] if label else [], source_chunks=[ci])

    return G
```

# 6. 엔티티 정규화

짧은 alias를 긴 이름으로 합친다.

```python
def resolve_entities(G: nx.DiGraph) -> nx.DiGraph:
    nodes = list(G.nodes)
    aliases = {}

    for short in nodes:
        for long in nodes:
            if short != long and len(short) < len(long) and short in long:
                aliases[short] = long
                break

    def resolve(n):
        seen = set()
        while n in aliases and n not in seen:
            seen.add(n)
            n = aliases[n]
        return n

    G2 = nx.DiGraph()
    for n in nodes:
        c = resolve(n)
        if c not in G2:
            G2.add_node(c, **G.nodes[n])

    for u, v, d in G.edges(data=True):
        cu, cv = resolve(u), resolve(v)
        if cu == cv:
            continue
        if G2.has_edge(cu, cv):
            continue
        G2.add_edge(cu, cv, **d)

    return G2
```

# 7. edge/node weight

같은 엔티티나 관계가 여러 chunk에서 나오면 weight를 높인다.

```python
def add_edge_weights(G: nx.DiGraph) -> nx.DiGraph:
    for u, v, d in G.edges(data=True):
        d["weight"] = len(d.get("source_chunks", []))
    for n, d in G.nodes(data=True):
        d["weight"] = len(d.get("source_chunks", []))
    return G
```

    communities = defaultdict(list)

# 10. 커뮤니티 요약

커뮤니티 내부 엔티티와 관계를 LLM에게 요약시킨다.

```python
SUMMARY_PROMPT = """아래는 지식 그래프의 한 커뮤니티에 속한 엔티티와 관계 목록입니다.
이 커뮤니티의 주제와 주요 사실을 4~6문장의 한국어 단락으로 요약해 주세요.

엔티티:
{entities}

관계:
{relations}"""
```

```python
def summarize_community(G: nx.DiGraph, nodes: list[str]) -> str:
    nodes_set = set(nodes)
    ent_lines = [f"- {n} ({G.nodes[n].get('type', '?')})" for n in nodes]
    rel_lines = []

    for u, v, d in G.edges(data=True):
        if u in nodes_set and v in nodes_set:
            for rel in (d.get("relations") or [""]):
                rel_lines.append(f"- {u} --[{rel}]--> {v}")

    prompt = SUMMARY_PROMPT.format(
        entities="\n".join(ent_lines) or "(없음)",
        relations="\n".join(rel_lines) or "(없음)",
    )
    return llm.invoke([HumanMessage(content=prompt)]).content
```

# 12. local_search

질문에 들어 있는 엔티티를 찾아 주변 hop과 원문 chunk를 모아 답변한다.

```python
def local_search(question, index, hops):
    G, chunks = index["graph"], index["chunks"]
    matched = [n for n in G.nodes if n in question]
    if not matched:
        return "질문에 해당되는 entity 없음. Global search를 시도해 보세요"

    nodes = set(matched)
    for _ in range(hops):
        new = set()
        for n in nodes:
            new |= set(G.successors(n)) | set(G.predecessors(n))
        nodes |= new

    chunk_ids = set()
    for n in nodes:
        chunk_ids |= set(G.nodes[n].get("source_chunks", []))

    chunk_text = "\n\n".join(f"[chunk {i}] {chunks[i]}" for i in sorted(chunk_ids))

    edge_lines = []
    for u, v, d in G.edges(data=True):
        if u in nodes and v in nodes:
            for rel in (d.get("relations") or [""]):
                edge_lines.append(f"({u}) --[{rel}]--> ({v})")

    sys = SystemMessage(content="주어진 컨텍스트만 사용해서 한국어로 답하세요. 관계 정보로 다중 홉 추론도 가능합니다.")
    user = HumanMessage(content=f"질문: {question}\n\n관계:\n{chr(10).join(edge_lines)}\n\n청크:\n{chunk_text}")
    return llm.invoke([sys, user]).content
```

# 13. global_search

전체 주제 질문은 커뮤니티 요약을 모두 종합해서 답한다.

```python
def global_search(question, index) -> str:
    all_summary_text = "\n\n".join(
        f"[C{cid}]\n{s}" for cid, s in sorted(index["summaries"].items())
    )
    sys = SystemMessage(content=(
        "당신은 지식 그래프의 커뮤니티 요약을 종합해 답하는 전문가입니다. "
        "모든 요약을 비교/종합해 한국어로 답하세요."
    ))
    user = HumanMessage(content=f"질문 : {question}\n\n== 커뮤니티 요약 ==\n{all_summary_text}")
    return llm.invoke([sys, user]).content
```

# 15. 자주 나오는 실수/주의사항

### 실수 1: source chunk를 저장하지 않음

```python
G.add_node(name, type=..., source_chunks=[ci])
```

> GraphRAG 답변에는 원문 근거가 필요하다.

### 실수 2: alias 정규화 후 edge 정보 손실

```text
같은 edge가 합쳐질 때 relations/source_chunks를 병합해야 한다.
```

### 실수 3: local_search에서 엔티티 매칭 실패

```python
matched = [n for n in G.nodes if n in question]
```

> 표현이 다르면 매칭이 안 되므로 실제 시스템에서는 entity linker가 필요하다.

### 실수 4: community 요약만으로 세부 질문 답변

```text
세부 질문은 local search, 전체 경향 질문은 global search가 적합하다.
```
