# 2. Tavily 검색 도구

### 설정 및 사용

```python
from langchain_community.tools.tavily_search import TavilySearchResults

# .env 에 TAVILY_API_KEY=tvly-dev-xxxx 필요
search_tool = TavilySearchResults(max_results=3)

search_tool.name         # → 'tavily_search_results_json'
search_tool.description  # → '...'
```

### 검색 실행

```python
results = search_tool.invoke({'query': '2026년 한국 ETF 시장 동향'})

for i, r in enumerate(results, 1):
    print(f"[{i}] {r.get('url', 'N/A')}")
    print(f"   {r.get('content', '')[:100]}...")
```

> 결과는 [{'url': ..., 'content': ...}, ...] 형태의 리스트

### LLM과 연결

```python
llm_with_search = llm.bind_tools([search_tool])

query    = "2026년 나스닥 전망은?"
messages = [HumanMessage(content=query)]
response = llm_with_search.invoke(messages)
messages.append(response)

if response.tool_calls:
    for tc in response.tool_calls:
        result      = search_tool.invoke(tc['args'])
        result_text = str(result)[:2000]    # 너무 길면 잘라서 전달
        messages.append(ToolMessage(content=result_text, tool_call_id=tc['id']))
    final = llm_with_search.invoke(messages)
    print(final.content)
```

### 결과 포매팅 함수

```python
def format_search_results(results, max_summary=100):
    formatted = []
    for r in results:
        content = r.get("content", "")
        summary = content[:max_summary] + "..." if len(content) > max_summary else content
        formatted.append({"url": r.get('url', ''), "summary": summary})
    return formatted
```

```python
def extract_key_sentences(text, keyword, max_sentences=3):
    sentences = [s.strip() for s in text.split(".") if s.strip()]
    matched   = [s + "." for s in sentences if keyword.lower() in s.lower()]
    return matched[:max_sentences]
```

# 5. 도구 성능 비교 실험

```python
import time

query = "2026 4월 AI 트렌드"
tools = {
    "Tavily"   : search_tool,
    "Wikipedia": wiki_tool,
    "DuckDuckGo": ddg_tool
}

for name, tool in tools.items():
    start = time.time()
    try:
        result    = tool.invoke({'query': query} if name == 'Tavily' else query)
        elapsed   = time.time() - start
        result_str = str(result)
        print(f"{name}: {elapsed:.2f}sec")
        print(f"결과 길이: {len(result_str)}")
        print(f"{result_str[:200]}")
    except:
        print(f"{name}: error")
    print("=======")
```

### 비교 결과 (일반적)

```text
Tavily     : 1~2초, 최신 정보, 구조화된 결과
Wikipedia  : 0.5~1초, 정확하지만 최신 정보 없음
DuckDuckGo : 0.5~1.5초, 무료, 품질 가변
```

# 6. 커스텀 도구 — calculator

```python
import math
from langchain_core.tools import tool

@tool
def calculator(expression):
    """math expression with eval"""
    try:
        allowed = {
            'math': math, 'abs': abs, 'round': round,
            'min': min, 'max': max
        }
        result = eval(expression, {"__builtins__": {}}, allowed)
        return str(result)
    except:
        return "calculation error"

calculator.invoke({'expression': 'math.sqrt(144) + math.pi'})
# → '12.141592653589793'
```

> `{"__builtins__": {}}` 로 위험한 내장 함수 접근 차단  
> `allowed` 딕셔너리에 허용할 함수만 명시적으로 등록

# 7. PythonREPL 도구

LLM이 직접 파이썬 코드를 생성하고 실행시킬 수 있다.

```python
from langchain_experimental.tools import PythonREPLTool

python_tool = PythonREPLTool()

python_tool.invoke("print(sum(range(1, 101)))")  # → 5050

# 복잡한 코드도 가능
code_str = """
import math
data = [23, 45, 12, 67, 34, 89, 56]
avg = sum(data) / len(data)
std = math.sqrt(sum((x - avg)**2 for x in data) / len(data))
print(f"평균: {avg:.1f}, 표준편차: {std:.1f}")
"""
python_tool.invoke(code_str)
```

### LLM + PythonREPL 파이프라인

```python
def run_code_agent(question, llm_with_tools, python_tool):
    messages = [HumanMessage(content=question)]
    response = llm_with_tools.invoke(messages)
    messages.append(response)

    code_text = ""
    output    = ""

    if response.tool_calls:
        for tc in response.tool_calls:
            code_text = tc['args'].get('query', '')
            output    = str(python_tool.invoke(tc['args']))
            messages.append(ToolMessage(content=output, tool_call_id=tc['id']))
        final = llm_with_tools.invoke(messages)
        return {"code": code_text, "output": output.strip(), "answer": final.content}

    return {"code": "", "output": "", "answer": response.content}

result = run_code_agent("1부터 100까지의 소수를 구해줘", llm_with_python, python_tool)
print(result['code'])    # LLM이 생성한 코드
print(result['output'])  # 실행 결과
print(result['answer'])  # LLM의 최종 답변
```

> PythonREPLTool은 편리하지만 보안상 위험  
> 프로덕션 환경에서는 샌드박스 격리 필수

# 8. 복합 에이전트 (agent_loop)

여러 도구를 하나의 에이전트에 등록하고 LLM이 상황에 맞는 도구를 선택하게 한다.

```python
all_tools = [search_tool, wiki_tool, ddg_tool, calculator]
llm_agent  = llm.bind_tools(all_tools)
tool_map   = {t.name: t for t in all_tools}

def agent_loop(query, llm_agent, tool_map, max_turns=3):
    messages   = [HumanMessage(content=query)]
    tools_used = []

    for turn in range(max_turns):
        response = llm_agent.invoke(messages)
        messages.append(response)

        if not response.tool_calls:
            return {'query': query, 'tools_used': tools_used}

        for tc in response.tool_calls:
            tool_name = tc['name']
            print(f"  [{turn+1}] {tool_name}({tc['args']})")
            tool_fn = tool_map.get(tool_name)
            if tool_fn:
                result = tool_fn.invoke(tc['args'])
            else:
                result = f"도구 {tool_name}를 찾을 수 없습니다"
            messages.append(ToolMessage(content=str(result)[:2000], tool_call_id=tc['id']))

    final = llm_agent.invoke(messages)
    return final.content
```

### 실행 예시

```python
test_queries = [
    '2026년 AI 스타트업 투자 동향을 알려줘',  # → Tavily 사용
    '랭체인이 뭔지 설명해줘',                  # → Wikipedia 사용
    '원주율 x 100의 제곱근은?'                # → calculator 사용
]

for q in test_queries:
    answer = agent_loop(q, llm_agent, tool_map)
    print(answer)
```

# 9. search_and_summarize — 검색 + 요약 파이프라인

```python
def search_and_summarize(query, search_tool, llm, max_results=3):
    raw_results = search_tool.invoke({'query': query})

    sources      = []
    combined_text = ""
    for r in raw_results[:max_results]:
        url     = r.get('url', 'no source')
        content = r.get('content', '')
        sources.append(url)
        combined_text += f"[출처: {url}]\n{content}\n\n"

    summary_prompt = f"""다음 검색 결과를 바탕으로 '{query}'에 대한 리포트를 작성하세요.

검색 결과:
{combined_text[:3000]}

리포트 형식:
1. 핵심 요약 (2~3문장)
2. 주요 포인트 (3개)
3. 출처 목록"""

    response = llm.invoke([HumanMessage(content=summary_prompt)])
    return {
        'query'           : query,
        'sources'         : sources,
        'report'          : response.content,
        'raw_result_count': len(raw_results)
    }
```

# 10. ToolWithFallback — 장애 대비 패턴

주 도구가 실패했을 때 대체 도구를 자동으로 사용한다.

```python
class ToolWithFallback:
    def __init__(self, primary, fallback=None, max_retries=2):
        self.primary    = primary
        self.fallback   = fallback
        self.max_retries = max_retries
        self.stats = {'primary_ok': 0, 'fallback_ok': 0, 'errors': 0}

    def invoke(self, args):
        for attempt in range(self.max_retries):
            try:
                result = self.primary.invoke(args)
                if result and str(result).strip():
                    self.stats['primary_ok'] += 1
                    return result
            except:
                print(f"{self.primary.name} error (attempt {attempt+1})")

        if self.fallback:
            try:
                fallback_args = args.get('query', args) if isinstance(args, dict) else args
                result = self.fallback.invoke(fallback_args)
                self.stats['fallback_ok'] += 1
                return f"[fallback: {self.fallback.name}] {result}"
            except:
                print("fallback fail too")

        self.stats['errors'] += 1
        return "no result"

# 사용
safe_search = ToolWithFallback(primary=search_tool, fallback=ddg_tool)
result = safe_search.invoke({"query": "랭체인 최신 업데이트"})
```

# 11. 핵심 개념 총정리

### 도구 선택 기준

```text
실시간 최신 정보가 필요   → Tavily
백과사전 수준의 설명 필요 → Wikipedia
무료 검색이 필요          → DuckDuckGo
수식 계산                 → calculator
코드 실행이 필요          → PythonREPL
```

### 전체 에이전트 흐름

```text
사용자 질문
    ↓
llm.bind_tools(all_tools).invoke(messages)
    ↓ tool_calls 있음
tool_map[name].invoke(args) → 결과
    ↓
ToolMessage(content=result, tool_call_id=id)
    ↓
llm.invoke(messages) → 최종 답변
```

### 실수 1: Tavily와 Wikipedia의 인자 형식 혼동

```python

result = search_tool.invoke({'query': '검색어'})

result = wiki_tool.invoke('검색어')
```

```python
# 올바른 코드 → 잘라서 전달
```
