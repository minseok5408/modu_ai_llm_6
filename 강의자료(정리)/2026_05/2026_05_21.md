# 1. 오늘 강의 개요

```text
PDF/HTML/DOCX/PPTX

partition
  ↓

  ↓
chunking
  ↓
Document 변환
  ↓

```

# 4. fast vs hi_res 전략

PDF 전용 파서에서는 전략을 명시할 수 있다.

```python
from unstructured.partition.pdf import partition_pdf

elements = partition_pdf(filename=pdf_path, strategy="fast")
elements_hi = partition_pdf(
    filename=pdf_path,
    strategy="hi_res",
    infer_table_structure=True,
)
```

```text
fast:
  - 텍스트 위주 추출
  - 빠름

hi_res:
  - 레이아웃, 표, 이미지 구조까지 더 자세히 분석
  - 느리지만 표/레이아웃 문서에 유리
```

```python
for c, n in Counter(el.category for el in elements).most_common():
    print(f" {c}: {n}")

for c, n in Counter(el.category for el in elements_hi).most_common():
    print(f" {c}: {n}")
```

# 5. 문서 분석 함수

요소 타입 분포, 페이지별 분포, 긴 본문 요소를 요약한다.

```python
from collections import defaultdict


def analyze_document(pdf_path: str, strategy: str = "fast") -> dict:
    els = partition_pdf(filename=pdf_path, strategy=strategy)
    type_stats = dict(Counter(el.category for el in els))
    page_dist = defaultdict(lambda: Counter())

    for el in els:
        p = int(el.metadata.page_number or 0)
        page_dist[p][el.category] += 1

    narratives = [el for el in els if el.category == "NarrativeText"]
    narratives.sort(key=lambda e: len(str(e)), reverse=True)
    top3 = [{
        "length": len(str(el)),
        "page": int(el.metadata.page_number or 0),
        "text_preview": str(el)[:80],
    } for el in narratives[:3]]

    return {
        "total_elements": len(els),
        "type_statistics": type_stats,
        "page_distribution": {p: dict(c) for p, c in page_dist.items()},
        "top3_longest_narratives": top3,
    }
```


### adaptive_chunk

```python
def adaptive_chunk(elements: list) -> list:
    if not elements:
        return []

    avg_len = sum(len(str(e)) for e in elements) / len(elements)
    if avg_len >= 200:
        max_chars = 1000
    elif avg_len >= 50:
        max_chars = 500
    else:
        max_chars = 300

    return chunk_by_title(
        elements,
        max_characters=max_chars,
        combine_text_under_n_chars=200,
    )
```

```python
for mode in ["single", "elements", "paged"]:
```

# 10. answer_from_pdf

PDF를 파싱하고, chunking, embedding, retrieval, generation까지 한 함수로 묶는다.

```python
from langchain_openai import ChatOpenAI, OpenAIEmbeddings
from langchain_community.vectorstores import FAISS
from langchain_core.messages import SystemMessage, HumanMessage
from langchain_core.documents import Document


def answer_from_pdf(pdf_path, question, k=3) -> dict:
    els = partition_pdf(filename=pdf_path, strategy="fast")
    cleaned = [el for el in els if el.category not in ["Header", "Footer", "PageBreak"]]
    chunks = chunk_by_title(
        cleaned,
        max_characters=600,
        combine_text_under_n_chars=150,
    )

    docs = [
        Document(
            page_content=str(c),
            metadata={
                "source": os.path.basename(pdf_path),
                "page": getattr(c.metadata, "page_number", None),
                "category": getattr(c.metadata, "category", None),
            },
        )
        for c in chunks
        if str(c).strip()
    ]

    embeddings = OpenAIEmbeddings(model="text-embedding-3-small")
    vectorstore = FAISS.from_documents(docs, embeddings)
    hits = vectorstore.similarity_search(question, k=k)
    context = "\n\n".join(
        f"[p{h.metadata.get('page', '?')}] {h.page_content}" for h in hits
    )

    llm = ChatOpenAI(model="gpt-4o-mini")
    msg = llm.invoke([
        SystemMessage(content="주어진 컨텍스트에 근거해 답변하세요"),
        HumanMessage(content=f"컨텍스트:\n{context}\n\n질문: {question}"),
    ])

    return {
        "answer": msg.content,
        "sources": [
            {"page": h.metadata.get("page", "?"), "preview": h.page_content[:30]}
            for h in hits
        ],
    }
```

# 11. HuggingFace SFTTrainer 보조 실습

[5월 20일 정리](2026_05_20.md)의 10절 Qwen LoRA 설정에 이어 `trl.SFTTrainer`로 학습한다. 새 노트북이라면 아래 항목을 먼저 준비한다.

- `model`: `Qwen/Qwen2.5-1.5B-Instruct`에 LoRA를 적용한 모델
- `tokenizer`: 같은 모델의 토크나이저와 채팅 템플릿
- `ds`: `messages` 열이 있는 JSONL을 불러온 `train` / `validation` 데이터셋
- 실행 환경: `torch`, `transformers`, `peft`, `datasets`, `trl`, `accelerate`가 설치된 환경. 아래 설정은 BF16을 지원하는 GPU 환경을 전제로 하며, 다른 환경에서는 모델 로딩 dtype과 `bf16` 설정을 함께 조정한다.

전날 생성한 train/validation 파일 경로를 확인하고 모델·토크나이저·데이터 로딩 셀을 먼저 실행한다. 라이브러리 버전에 따라 인자가 달라질 수 있으므로 사용한 버전을 기록한다. ([SFTTrainer 공식 문서](https://huggingface.co/docs/trl/sft_trainer))

```python
from trl import SFTTrainer, SFTConfig

OUTPUT_DIR = "qwen25-1.5b_polite_email_lora"

sft_cfg = SFTConfig(
    output_dir=OUTPUT_DIR,
    num_train_epochs=80,
    per_device_train_batch_size=2,
    per_device_eval_batch_size=2,
    learning_rate=2e-4,
    warmup_ratio=0.1,
    lr_scheduler_type="cosine",
    bf16=True,
    max_length=512,
)

trainer = SFTTrainer(
    model=model,
    args=sft_cfg,
    train_dataset=ds["train"],
    eval_dataset=ds["validation"],
    processing_class=tokenizer,
)
trainer.train()

# 학습 종료 후 validation 데이터로 평가
metrics = trainer.evaluate()
print(metrics)
```

`num_train_epochs=80`은 강의 실험값이며 모든 데이터에 권장되는 횟수는 아니다. `per_device_train_batch_size=2`는 장치당 배치 크기이고, `max_length=512`는 학습 시퀀스 길이 제한이다. 긴 예제가 잘려 답변 부분이 사라지지 않는지 토큰화 결과를 확인한다.

`trainer.evaluate()`의 결과에서 `eval_loss`를 확인한다. 이는 검증 데이터의 다음 토큰 예측 손실이며 답변 정확도 자체는 아니다. 동일한 검증 데이터와 설정에서 손실과 생성 답변을 함께 비교한다. 학습 중 검증 손실 추이를 보려면 `eval_strategy`를 별도로 설정해야 한다.
